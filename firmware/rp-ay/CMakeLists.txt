# Прошивка RP-AY макета Тринити: RP в разрыве слота YM2203 на MultiSound.
# Сборка - build.bat (пути к SDK и инструментам - в нём).
cmake_minimum_required(VERSION 3.13)

set(PICO_PLATFORM rp2350-arm-s)       # RP2350, ядра ARM
set(PICO_BOARD spotpear_rp2350b_mini)
set(PICO_BOARD_HEADER_DIRS ${CMAKE_CURRENT_LIST_DIR}/boards)

include(pico_sdk_import.cmake)
project(rp_ay C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()

add_executable(rp_ay main.c config.c modes.c chip.c card.c ../../engine/ayfx.c ../../engine/svc.c)   # движок AY FX и API - общие (engine/)
target_include_directories(rp_ay PRIVATE ${CMAKE_CURRENT_LIST_DIR}/../../engine)
pico_generate_pio_header(rp_ay ${CMAKE_CURRENT_LIST_DIR}/ws2812.pio)
option(BOARD_REV_A "Плата рев А, иначе макет" OFF)
option(BUS_SELFTEST "Самотест шины, только на голой RP" OFF)
if(BOARD_REV_A)
    target_compile_definitions(rp_ay PRIVATE BOARD_REV_A)
    pico_generate_pio_header(rp_ay ${CMAKE_CURRENT_LIST_DIR}/bus_ay.pio)
else()
    pico_generate_pio_header(rp_ay ${CMAKE_CURRENT_LIST_DIR}/bus.pio)
endif()
if(BUS_SELFTEST)
    target_compile_definitions(rp_ay PRIVATE BUS_SELFTEST)
endif()
option(SNDH "SNDH-плеер (Atari, doc/trinity-sndh.md)" OFF)
if(SNDH)
    set(AA ${CMAKE_CURRENT_LIST_DIR}/../../engine/third_party/atariaudio)
    file(GLOB AA_SRC ${AA}/*.cpp ${AA}/external/*.cpp ${AA}/external/Musashi/*.cpp)
    target_sources(rp_ay PRIVATE ../../engine/sndh/sndh.cpp ${AA_SRC})
    target_include_directories(rp_ay PRIVATE ${AA} ${CMAKE_CURRENT_LIST_DIR}/../../engine/sndh)
    target_compile_definitions(rp_ay PRIVATE SNDH NDEBUG TRINITY_M68K_COMPACT)
endif()
target_link_libraries(rp_ay pico_stdlib pico_multicore hardware_pio hardware_gpio hardware_clocks hardware_vreg)
pico_enable_stdio_usb(rp_ay 1)      # консоль - по USB (COM-порт)
pico_enable_stdio_uart(rp_ay 0)
pico_set_binary_type(rp_ay copy_to_ram)   # вся программа - в ОЗУ: без промахов кэша флеша (разбор шины успевает)
target_link_options(rp_ay PRIVATE -Wl,--print-memory-usage)
pico_add_extra_outputs(rp_ay)       # rp_ay.uf2 - кинуть на диск RP (BOOT + USB)
